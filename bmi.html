โค้ดฝั่ง Google Apps Script (Code.gs)

คัดลอกโค้ดนี้ไปวางในไฟล์ Code.gs ของ Google Sheets

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // สร้าง Header เมื่อใช้ชีตใหม่ครั้งแรก
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["วัน-เวลา", "ชื่อ-นามสกุล", "อายุ", "เพศ", "น้ำหนัก (kg)", "ส่วนสูง (cm)", "ค่า BMI", "ผลการประเมิน"]);
    }
    
    // บันทึกข้อมูลแถวใหม่
    sheet.appendRow([
      data.timestamp || new Date().toLocaleString("th-TH"),
      data.name,
      data.age || "-",
      data.gender || "-",
      data.weight,
      data.height,
      data.bmi,
      data.category
    ]);
    
    return ContentService
      .createTextOutput(JSON.stringify({ status: "success", message: "บันทึกเรียบร้อย" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
